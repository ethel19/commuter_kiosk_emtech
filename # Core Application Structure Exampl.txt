class CashMachine:

  def __init__(self, initial_balance=5000.0):
    self.balance = initial_balance

  def calculate_coin_exchange(self, bill_amount: float, service_fee_rate=0.0):
    fee = bill_amount * service_fee_rate
    net_amount = bill_amount - fee
    # Logic to break net_amount into Philippine coin denominations (20, 10, 5, 1)
    return {"net_amount": net_amount, "fee": fee}


class PhilippineFareCalculator:

  def compute_fare(
    self, distance_km: float, vehicle_type: str, is_discounted: bool = False
  ):
    # Example logic reflecting LTFRB distance-add-on frameworks
    base_fare = 13.00  # Example base fare for first 5km
    succeeding_rate = 2.25  # Example per km rate

    if distance_km <= 5:
      fare = base_fare
    else:
      fare = base_fare + ((distance_km - 5) * succeeding_rate)

    if is_discounted:
      fare *= 0.80  # 20% discount for students/seniors/PWDs

    return round(fare, 2)
